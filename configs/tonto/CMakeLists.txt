cmake_minimum_required(VERSION 3.20)
project(tonto_ir LANGUAGES Fortran)
get_filename_component(STUDY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SOURCE_ROOT "${STUDY_ROOT}/spec-cpu2006-redist/modified/465.tonto/src")
file(GLOB SOURCES "${SOURCE_ROOT}/*.F90" "${SOURCE_ROOT}/*.f90")
add_library(tonto_ir OBJECT ${SOURCES})
target_include_directories(tonto_ir PRIVATE "${SOURCE_ROOT}")
target_compile_definitions(tonto_ir PRIVATE SPEC_CPU SPEC_CPU_LP64 USE_PRE_AND_POST_CONDITIONS)
target_compile_options(tonto_ir PRIVATE -O0 -emit-llvm)
