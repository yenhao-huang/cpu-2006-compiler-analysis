cmake_minimum_required(VERSION 3.20)
project(wrf_ir LANGUAGES C Fortran)
get_filename_component(STUDY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SOURCE_ROOT "${STUDY_ROOT}/spec-cpu2006-redist/modified/481.wrf/src")
file(GLOB SOURCES "${SOURCE_ROOT}/*.F90" "${SOURCE_ROOT}/*.f90" "${SOURCE_ROOT}/*.c" "${SOURCE_ROOT}/netcdf/*.c")
add_library(wrf_ir OBJECT ${SOURCES})
target_include_directories(wrf_ir PRIVATE "${SOURCE_ROOT}" "${SOURCE_ROOT}/netcdf/include")
target_compile_definitions(wrf_ir PRIVATE SPEC_CPU SPEC_CPU_LINUX SPEC_CPU_LP64 SPEC_CPU_CASE_FLAG IWORDSIZE=4 DWORDSIZE=8 RWORDSIZE=4 LWORDSIZE=4 EM_CORE=1 NMM_CORE=0 COAMPS_CORE=0 DA_CORE=0 NETCDF INTIO MAX_DOMAINS_F=21 CONFIG_BUF_LEN=8192 F90_STANDALONE)
target_compile_options(wrf_ir PRIVATE "$<$<COMPILE_LANGUAGE:Fortran>:-O0;-emit-llvm;-cpp>" "$<$<COMPILE_LANGUAGE:C>:-std=gnu89;-O3;-Xclang;-disable-llvm-passes;-emit-llvm;-fcommon;-fno-strict-aliasing;-Wno-error=implicit-function-declaration;-Wno-error=int-conversion>")
